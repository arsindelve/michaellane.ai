export async function getStaticPaths() {
  const all = (await getCollection('work')).sort((a, b) => a.data.order - b.data.order);

    params: { slug: entry.id },
    props: { entry, prev: all[(i - 1 + all.length) % all.length], next: all[(i + 1) % all.length] },
  }));
}

const esc = (t: string) => t.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');

// A tiny single-pass tokenizer per language. Each rule is tried at the current position; the
// first match wins and is wrapped once, so one rule can never re-color another rule's markup.
type Rule = [cls: string, re: RegExp];
const words = (list: string, flags = '') => new RegExp(`\\b(?:${list.split(' ').join('|')})\\b`, `y${flags}`);
const RULES: Record<'basic' | 'cs' | 'py' | 'css', Rule[]> = {
  basic: [
    ['cm', /'.*$/y],
    ['st', /"[^"]*"?/y],
    ['kw', words('DO UNTIL LOOP IF THEN ELSEIF ELSE END LINE SCREEN AND OR BF SUB DIM SHARED FOR TO NEXT PRINT LOCATE COLOR INPUT OPEN CLOSE AS GOSUB RETURN CALL SELECT CASE DECLARE COMMON DATA')],
    ['fn', /[A-Z][A-Z0-9]*\$?(?=\()/y],
    ['nu', /\b\d+\b/y],
  ],
  cs: [
    ['cm', /\/\/.*$/y],
    ['st', /\$?"(?:[^"\\]|\\.)*"?|'(?:[^'\\]|\\.)*'|`[^`]*`?/y],
    ['kw', words('private public protected internal static readonly async await var let const function export import from new return if else foreach for while switch case break continue null true false is not and or in out string bool int void class using this throw try catch finally default when interface type as')],
    ['fn', /[A-Za-z_][A-Za-z0-9_]*(?=\s*[(<])/y],
    ['nu', /\b\d+\b/y],
  ],
  py: [
    ['cm', /#.*$/y],
    ['st', /"""[^]*?(?:"""|$)|'''[^]*?(?:'''|$)|[rbfu]?"(?:[^"\\]|\\.)*"?|[rbfu]?'(?:[^'\\]|\\.)*'?/y],
    ['kw', words('def class return if elif else for while in not and or is None True False import from as with try except finally raise lambda yield async await pass continue break self')],
    ['fn', /[A-Za-z_][A-Za-z0-9_]*(?=\()/y],
    ['nu', /\b\d+\b/y],
  ],
  css: [
    ['cm', /\/\*.*?(?:\*\/|$)/y],
    ['kw', /^[^{}]+(?=\{)/y],
    ['fn', /[a-z-]+(?=\s*:[^;{}]*;)/y],
    ['st', /#[0-9a-fA-F]{3,8}\b/y],
    ['nu', /\b\d*\.?\d+(?:px|rem|em|vw|vh|%|s|deg)?\b/y],
  ],
};
const highlight = (line: string, lang: keyof typeof RULES) => {
  const rules = RULES[lang];
  let out = '';
  let plain = '';
  for (let i = 0; i < line.length; ) {
    let hit = '';
    let cls = '';
    for (const [c, re] of rules) {
      re.lastIndex = i;
      const m = re.exec(line);
      if (m && m[0]) {
        hit = m[0];
        cls = c;
        break;
      }
    }
    if (hit) {
      out += esc(plain) + `<span class="${cls}">${esc(hit)}</span>`;
      plain = '';
      i += hit.length;
    } else {
      plain += line[i++];
    }
  }
  return out + esc(plain);
};

      })
      .join('')
